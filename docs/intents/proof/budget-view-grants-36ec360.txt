DEFINER = oteru_budget_view_definer SQL SECURITY DEFINER
AS WITH
    fct AS
    (
        SELECT
            day_utc,
            period_start,
            scope,
            scope_key,
            team,
            project,
            owner_team,
            budget_usd_micros,
            consumed_period_usd_micros,
            consumed_milli,
            warn_milli,
            breach_milli,
            budget_status,
            action_on_breach,
            last_cost_ts
        FROM otel.fct_budget_consumption_daily
    ),
    wm AS
    (
        SELECT
            max(day_utc) AS last_day,
            max(last_cost_ts) AS as_of_ts
        FROM fct
    ),
    key_last AS
    (
        SELECT
            scope,
            scope_key,
            max(day_utc) AS key_last_day
        FROM fct
        GROUP BY
            scope,
            scope_key
    )
SELECT
    CAST(f.scope, 'LowCardinality(String)') AS scope,
    CAST(f.scope_key, 'String') AS scope_key,
    CAST(f.team, 'String') AS team,
    CAST(f.project, 'String') AS project,
    CAST(f.owner_team, 'String') AS owner_team,
    CAST(f.period_start, 'Date') AS period_start,
    CAST(f.day_utc, 'Date') AS as_of_day,
    CAST(w.as_of_ts, 'DateTime64(9)') AS as_of_ts,
    CAST(f.budget_usd_micros, 'Int64') AS budget_usd_micros,
    CAST(f.consumed_period_usd_micros, 'Int64') AS consumed_period_usd_micros,
    CAST(f.consumed_milli, 'Int64') AS consumed_milli,
    CAST(f.warn_milli, 'UInt16') AS warn_milli,
    CAST(f.breach_milli, 'UInt16') AS breach_milli,
    CAST(f.budget_status, 'LowCardinality(String)') AS budget_status,
    CAST(f.action_on_breach, 'LowCardinality(String)') AS action_on_breach,
    CAST(26, 'UInt16') AS stale_threshold_hours,
    CAST(w.as_of_ts < (now64(9, 'UTC') - toIntervalHour(26)), 'Bool') AS is_stale
FROM fct AS f
INNER JOIN key_last AS k ON (f.scope = k.scope) AND (f.scope_key = k.scope_key) AND (f.day_utc = k.key_last_day)
